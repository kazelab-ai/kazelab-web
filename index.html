use anthropic_sdk::prelude::*;
use kazelab_synapse::{AgentMesh, McpRegistry, Policy};

#[tokio::main]
async fn main() -> Result<()> {
    // Connect to Claude 3.5 Sonnet Brain
    let brain = Anthropic::new()
        .model("claude-3-5-sonnet-20241022")
        .cache_prompt(true)
        .temperature(0.0);

    let mut mesh = AgentMesh::bootstrap(brain).await?;
    mesh.register_mcp_cluster("mcp://cluster.kazelab.xyz").await?;

    // Self-directed reasoning & verification loop
    let outcome = mesh.execute(Goal {
        target_repo: "org/payment-service",
        constraint: Policy::ZeroDefectFormalProof,
    }).await?;

    println!("Verified zero-defect patch converged in {}ms", outcome.duration_ms);
    Ok(())
}
